        ToolTip tp = new ToolTip();

        {
            InitializeComponent();
            listBox1.Items.AddRange(new object[] {
            "item1", "item2", "item3", 1, "item4"
            });
            tp.SetToolTip(button1, "Click if you want to clear list");
            tp.SetToolTip(button2, "Click if you want to clear item");
            tp.SetToolTip(button3, "Click if you want to copy list");
        }

        {

        }

        private void listBox1_SelectedIndexChanged(object sender, EventArgs e)
        {

        }

        private void label1_Click(object sender, EventArgs e)
        {

        }

        private void button1_Click(object sender, EventArgs e)
        {
            listBox1.Items.Clear();
        }

        private void button2_Click(object sender, EventArgs e)
        {
            if (listBox1.Items.Count != 0)
            {
                for (int i = 0; i < listBox1.Items.Count; i++)
                {
                    listBox1.Items.Remove(listBox1.SelectedItems[i]);
                }
                MessageBox.Show("You delete something");
            }
        }

        private void richTextBox1_TextChanged(object sender, EventArgs e)
        {

        }

        private void button3_Click(object sender, EventArgs e)
        {
            richTextBox1.Clear();

            foreach (var item in listBox1.Items)
            {
                richTextBox1.AppendText(item.ToString() + Environment.NewLine);
            }
        }

        private void button4_Click(object sender, EventArgs e)
        {
            progressBar1.Minimum = 0;
            progressBar1.Maximum = 700;
            progressBar1.Step = 1;
            for (int i = 0; i <= 700; i++)
            {
                progressBar1.PerformStep();
                label2.Text = "Value: " + progressBar1.Value.ToString();
                Update();
                Thread.Sleep(100);
            }
        }

        private void button5_Click(object sender, EventArgs e)
        {
            Color color = Color.FromArgb(trackBar1.Value, trackBar2.Value, trackBar3.Value);
            button5.BackColor = color;
        }

        private void button6_Click(object sender, EventArgs e)
        {
            Color color1 = Color.FromArgb((int)numericUpDown1.Value, (int)numericUpDown2.Value, (int)numericUpDown3.Value);
            listBox1.BackColor = color1;
        }
    }
}
